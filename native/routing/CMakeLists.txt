cmake_minimum_required(VERSION 3.24)
project(ZSDeskRouting LANGUAGES C CXX RC)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "ZSDesk routing requires Windows x64")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(MINHOOK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../crates/zsdesk-windows/native/minhook")
add_library(ZSDeskRevealRouter SHARED adapter/ZSDeskRevealRouter.cpp
  "${MINHOOK_ROOT}/src/buffer.c" "${MINHOOK_ROOT}/src/hook.c"
  "${MINHOOK_ROOT}/src/trampoline.c" "${MINHOOK_ROOT}/src/hde/hde64.c")
target_include_directories(ZSDeskRevealRouter PRIVATE "${MINHOOK_ROOT}/include")
target_compile_definitions(ZSDeskRevealRouter PRIVATE UNICODE _UNICODE)
target_compile_options(ZSDeskRevealRouter PRIVATE /utf-8 /W4 /WX /wd4505)
target_compile_options(ZSDeskRevealRouter PRIVATE "$<$<COMPILE_LANGUAGE:C>:/wd4201;/wd4310;/wd4244;/wd4701>")
target_link_libraries(ZSDeskRevealRouter PRIVATE user32 shell32 ole32 psapi advapi32 bcrypt)
target_link_options(ZSDeskRevealRouter PRIVATE "/PDBALTPATH:%_PDB%")
add_executable(ShellRoutingControl ShellRoutingControl.cpp controller.rc)
target_compile_definitions(ShellRoutingControl PRIVATE UNICODE _UNICODE)
target_compile_options(ShellRoutingControl PRIVATE /utf-8 /W4 /WX /wd4505)
target_link_options(ShellRoutingControl PRIVATE /SUBSYSTEM:WINDOWS /ENTRY:wmainCRTStartup "/PDBALTPATH:%_PDB%")
target_link_libraries(ShellRoutingControl PRIVATE user32 advapi32 shell32 ole32 bcrypt shlwapi version)
